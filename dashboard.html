function doGet(e) { var sheet = SpreadsheetApp.openById('1Ft1s1ZCh1o24JLmmMcqR5lpxnqB0i3wQl7HeKhmSYWk').getActiveSheet(); var p = e.parameter; if(p.action === 'read') { var rows = sheet.getDataRange().getValues(); rows.shift(); return ContentService.createTextOutput(JSON.stringify({status:'ok',rows:rows})).setMimeType(ContentService.MimeType.JSON); } if(p.action === 'statut') { var rows = sheet.getDataRange().getValues(); for(var i = 1; i < rows.length; i++) { if(String(rows[i][2]) === String(p.numero)) { sheet.getRange(i+1, 7).setValue(p.statut); return ContentService.createTextOutput(JSON.stringify({status:'ok'})).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({status:'error',message:'Commande non trouvee'})).setMimeType(ContentService.MimeType.JSON); } var lastRow = sheet.getLastRow(); var numero = String(lastRow).padStart(3,'0'); var maintenant = new Date(); var heureLocale = new Date(maintenant.getTime() + 60*60*1000); sheet.appendRow([heureLocale, p.prenom||'', numero, p.articles||'', p.details||'', p.total||'', 'attente']); return ContentService.createTextOutput(JSON.stringify({status:'ok',numero:numero})).setMimeType(ContentService.MimeType.JSON); }
